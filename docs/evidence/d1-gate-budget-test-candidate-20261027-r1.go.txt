func gateBudgetFixture(state GateState) gateBudgetSnapshot {
	now := time.Date(2026, 10, 27, 12, 0, 0, 250000000, time.UTC)
	tm := func(t time.Time) sql.NullTime { return sql.NullTime{Time:t, Valid:true} }
	s := gateBudgetSnapshot{Now:now, StateSchema:"system-backup-installation-state.v1", Installation:"installation_primary",
		ActiveVersion:8, Matrix:"k3s-single-node", ConfigSchema:"system-backup-config.v1", ConfigInstallation:"installation_primary", ConfigVersion:8,
		CurrentAcquisition:60, CurrentHold:600, GateInstallation:"installation_primary", Scope:CaptureGateScope, LockName:"system_backup_capture_gate",
		State:state, Generation:4, Fence:0, RowVersion:7, LockTTL:900}
	if state == GateIdle { return s }
	s.Owner = sql.NullString{String:"controller:gate-1", Valid:true}
	s.Acquisition = sql.NullInt64{Int64:30, Valid:true}
	s.AcquiringAt = tm(now.Add(-25*time.Second))
	s.AcquisitionDeadline = tm(s.AcquiringAt.Time.Add(30*time.Second))
	s.Heartbeat = tm(now.Add(-10*time.Second))
	s.LeaseExpires = s.AcquisitionDeadline
	if state == GateAcquiring { return s }
	s.Fence = 9
	s.Hold = sql.NullInt64{Int64:300, Valid:true}
	s.HeldAt = tm(now.Add(-250*time.Second))
	s.AcquiringAt = tm(s.HeldAt.Time.Add(-10*time.Second))
	s.AcquisitionDeadline = tm(s.AcquiringAt.Time.Add(30*time.Second))
	s.HoldDeadline = tm(s.HeldAt.Time.Add(300*time.Second))
	s.LeaseExpires = s.HoldDeadline
	if state == GateReleasing {
		s.ReleaseAt = tm(now.Add(-5*time.Second))
		s.Heartbeat = tm(now.Add(-4*time.Second))
		s.LeaseExpires = tm(now.Add(60*time.Second))
	}
	return s
}

func TestGateBudgetFrozenThresholdsAcrossConfigurationChanges(t *testing.T) {
	for _, state := range []GateState{GateIdle,GateAcquiring,GateHeld,GateReleasing} {
		t.Run(string(state),func(t *testing.T){
			s := gateBudgetFixture(state)
			p,err := projectGateBudget(s,s.Installation)
			if err != nil { t.Fatal(err) }
			if p.Profile!="single-node" || p.Runner!="k3s" { t.Fatalf("wrong original matrix projection: %+v",p) }
			wantWait,wantHold,wantAcq,wantMax := 0.0,0.0,60.0,600.0
			if state==GateAcquiring { wantWait,wantAcq=25,30 }
			if state==GateHeld || state==GateReleasing { wantHold,wantAcq,wantMax=250,30,300 }
			if p.WaitSeconds!=wantWait || p.HoldSeconds!=wantHold || p.AcquisitionTimeoutSeconds!=wantAcq || p.MaxHoldSeconds!=wantMax { t.Fatalf("current config replaced active immutable timing: %+v",p) }
			body,err := renderGateBudgetMetrics(p)
			if err!=nil || strings.Count(string(body),"# TYPE ")!=4 || strings.Contains(string(body),"installation_primary") || strings.Contains(string(body),s.Owner.String) && s.Owner.Valid { t.Fatalf("invalid exposition: %s, %v",body,err) }
		})
	}
}

func TestGateBudgetExpiredLeaseDoesNotHideOverdueGate(t *testing.T) {
	for _, state := range []GateState{GateAcquiring,GateHeld,GateReleasing} {
		t.Run(string(state),func(t *testing.T){
			s:=gateBudgetFixture(state)
			if state==GateAcquiring {
				s.AcquiringAt.Time=s.Now.Add(-40*time.Second)
				s.AcquisitionDeadline.Time=s.AcquiringAt.Time.Add(30*time.Second)
				s.Heartbeat.Time=s.AcquiringAt.Time.Add(5*time.Second)
				s.LeaseExpires=s.AcquisitionDeadline
			} else {
				s.HeldAt.Time=s.Now.Add(-350*time.Second)
				s.AcquiringAt.Time=s.HeldAt.Time.Add(-10*time.Second)
				s.AcquisitionDeadline.Time=s.AcquiringAt.Time.Add(30*time.Second)
				s.HoldDeadline.Time=s.HeldAt.Time.Add(300*time.Second)
				s.Heartbeat.Time=s.HeldAt.Time.Add(150*time.Second)
				s.LeaseExpires=s.HoldDeadline
				if state==GateReleasing { s.ReleaseAt.Time=s.Now.Add(-100*time.Second);s.Heartbeat.Time=s.Now.Add(-90*time.Second) }
			}
			p,err:=projectGateBudget(s,s.Installation)
			if err!=nil { t.Fatal(err) }
			if state==GateAcquiring && (p.WaitSeconds!=40 || p.AcquisitionTimeoutSeconds!=30) { t.Fatalf("overdue wait hidden: %+v",p) }
			if state!=GateAcquiring && (p.HoldSeconds!=350 || p.MaxHoldSeconds!=300) { t.Fatalf("overdue hold hidden or restarted at release: %+v",p) }
		})
	}
}

func TestGateBudgetExactBoundariesAndOriginalMatrixLabels(t *testing.T) {
	for _, entry:=range []struct{matrix,profile,runner string}{{"k8s-cluster","cluster","k8s"},{"k3s-cluster","cluster","k3s"},{"k8s-single-node","single-node","k8s"},{"k3s-single-node","single-node","k3s"}} {
		s:=gateBudgetFixture(GateIdle);s.Matrix=entry.matrix
		p,err:=projectGateBudget(s,s.Installation)
		if err!=nil || p.Profile!=entry.profile || p.Runner!=entry.runner { t.Fatalf("matrix %s: %+v %v",entry.matrix,p,err) }
	}
	for _, age:=range []time.Duration{0,239*time.Second+999999*time.Microsecond,240*time.Second,300*time.Second,301*time.Second} {
		s:=gateBudgetFixture(GateHeld)
		s.HeldAt.Time=s.Now.Add(-age);s.AcquiringAt.Time=s.HeldAt.Time.Add(-10*time.Second)
		s.AcquisitionDeadline.Time=s.AcquiringAt.Time.Add(30*time.Second);s.HoldDeadline.Time=s.HeldAt.Time.Add(300*time.Second)
		s.Heartbeat.Time=s.HeldAt.Time;s.LeaseExpires=s.HoldDeadline
		p,err:=projectGateBudget(s,s.Installation)
		if err!=nil || math.Abs(p.HoldSeconds-age.Seconds())>1e-9 { t.Fatalf("boundary %s: %+v %v",age,p,err) }
		if (p.HoldSeconds>=p.MaxHoldSeconds*.8)!=(age>=240*time.Second) || (p.HoldSeconds>=p.MaxHoldSeconds)!=(age>=300*time.Second) { t.Fatalf("incorrect original >= boundary: %+v",p) }
	}
}

func TestGateBudgetRejectsUnknownOrContradictoryOriginalFacts(t *testing.T) {
	cases:=map[string]func(*gateBudgetSnapshot){
		"wrong_installation":func(s *gateBudgetSnapshot){s.GateInstallation="installation_other"},
		"wrong_config_origin":func(s *gateBudgetSnapshot){s.ConfigInstallation="installation_other"},
		"wrong_scope":func(s *gateBudgetSnapshot){s.Scope="other"},
		"unknown_matrix":func(s *gateBudgetSnapshot){s.Matrix="docker-cluster"},
		"unknown_state":func(s *gateBudgetSnapshot){s.State="unknown"},
		"clock_offset":func(s *gateBudgetSnapshot){s.OffsetMicros=1},
		"timezone":func(s *gateBudgetSnapshot){s.Now=s.Now.In(time.FixedZone("other",3600))},
		"config_version":func(s *gateBudgetSnapshot){s.ConfigVersion++},
		"missing_frozen_budget":func(s *gateBudgetSnapshot){s.Hold.Valid=false},
		"expired_window_not_sane":func(s *gateBudgetSnapshot){s.HoldDeadline.Time=s.HeldAt.Time},
		"window_exceeds_frozen_budget":func(s *gateBudgetSnapshot){s.HoldDeadline.Time=s.HeldAt.Time.Add(301*time.Second)},
		"future_start":func(s *gateBudgetSnapshot){s.HeldAt.Time=s.Now.Add(time.Second)},
		"no_original_fence":func(s *gateBudgetSnapshot){s.Fence=0},
		"no_original_owner":func(s *gateBudgetSnapshot){s.Owner.Valid=false},
		"lease_after_absolute_hold":func(s *gateBudgetSnapshot){s.LeaseExpires.Time=s.HoldDeadline.Time.Add(time.Second)},
		"unexpected_release":func(s *gateBudgetSnapshot){s.ReleaseAt=sql.NullTime{Time:s.Now,Valid:true}},
	}
	for name,change:=range cases { t.Run(name,func(t *testing.T){s:=gateBudgetFixture(GateHeld);change(&s);if p,err:=projectGateBudget(s,s.Installation);err==nil || p!=(GateBudgetProjection{}) {t.Fatalf("unknown facts became healthy: %+v %v",p,err)}}) }
	idle:=gateBudgetFixture(GateIdle);idle.Hold=sql.NullInt64{Int64:300,Valid:true}
	if _,err:=projectGateBudget(idle,idle.Installation);err==nil {t.Fatal("idle accepted leftover active facts")}
}

type gateBudgetSQLScript struct {
	rows [][]driver.Value
	queryErr, readErr, closeErr, commitErr error
	onCommit func()
	begins,commits,rollbacks atomic.Int64
}
type gateBudgetSQLDriver struct{s *gateBudgetSQLScript}
type gateBudgetSQLConn struct{s *gateBudgetSQLScript}
type gateBudgetSQLTx struct{s *gateBudgetSQLScript}
type gateBudgetSQLRows struct{s *gateBudgetSQLScript;at int}
var gateBudgetSQLDriverID atomic.Uint64
func(d gateBudgetSQLDriver)Open(string)(driver.Conn,error){return &gateBudgetSQLConn{s:d.s},nil}
func(c *gateBudgetSQLConn)Prepare(string)(driver.Stmt,error){return nil,errors.New("unexpected prepare")}
func(c *gateBudgetSQLConn)Close()error{return nil}
func(c *gateBudgetSQLConn)Begin()(driver.Tx,error){return nil,errors.New("expected explicit read-only RR transaction")}
func(c *gateBudgetSQLConn)BeginTx(_ context.Context,o driver.TxOptions)(driver.Tx,error){if !o.ReadOnly || o.Isolation!=driver.IsolationLevel(sql.LevelRepeatableRead){return nil,errors.New("not read-only RR")};c.s.begins.Add(1);return &gateBudgetSQLTx{s:c.s},nil}
func(c *gateBudgetSQLConn)QueryContext(_ context.Context,q string,args []driver.NamedValue)(driver.Rows,error){if q!=gateBudgetSnapshotSQL || len(args)!=2 || args[0].Value!=CaptureGateScope || args[1].Value!="installation_primary" {return nil,errors.New("unexpected SQL source, bound installation or scope")};if c.s.queryErr!=nil{return nil,c.s.queryErr};return &gateBudgetSQLRows{s:c.s},nil}
func(t *gateBudgetSQLTx)Commit()error{t.s.commits.Add(1);if t.s.onCommit!=nil{t.s.onCommit()};return t.s.commitErr}
func(t *gateBudgetSQLTx)Rollback()error{t.s.rollbacks.Add(1);return nil}
func(r *gateBudgetSQLRows)Columns()[]string{cols:=make([]string,29);for i:=range cols{cols[i]=fmt.Sprint(i)};return cols}
func(r *gateBudgetSQLRows)Close()error{return r.s.closeErr}
func(r *gateBudgetSQLRows)Next(dst []driver.Value)error{if r.at>=len(r.s.rows){if r.s.readErr!=nil{return r.s.readErr};return io.EOF};copy(dst,r.s.rows[r.at]);r.at++;return nil}
func gateBudgetDriverRow(s gateBudgetSnapshot)[]driver.Value{
	nullTime:=func(v sql.NullTime)driver.Value{if v.Valid{return v.Time};return nil}
	nullInt:=func(v sql.NullInt64)driver.Value{if v.Valid{return v.Int64};return nil}
	var owner driver.Value;if s.Owner.Valid{owner=s.Owner.String}
	return []driver.Value{s.Now,s.OffsetMicros,s.StateSchema,s.Installation,int64(s.ActiveVersion),s.Matrix,s.ConfigSchema,s.ConfigInstallation,int64(s.ConfigVersion),s.CurrentAcquisition,s.CurrentHold,s.GateInstallation,s.Scope,s.LockName,string(s.State),int64(s.Generation),int64(s.Fence),int64(s.RowVersion),owner,s.LockTTL,nullTime(s.Heartbeat),nullTime(s.LeaseExpires),nullInt(s.Acquisition),nullInt(s.Hold),nullTime(s.AcquiringAt),nullTime(s.HeldAt),nullTime(s.ReleaseAt),nullTime(s.AcquisitionDeadline),nullTime(s.HoldDeadline)}
}
func openGateBudgetScript(t *testing.T,s *gateBudgetSQLScript)*sql.DB{t.Helper();name:=fmt.Sprintf("gate-budget-script-%d",gateBudgetSQLDriverID.Add(1));sql.Register(name,gateBudgetSQLDriver{s:s});db,err:=sql.Open(name,"");if err!=nil{t.Fatal(err)};t.Cleanup(func(){_ = db.Close()});return db}

func TestSQLGateBudgetReadsBoundedOriginalSourcesInOneUTCReadOnlySnapshot(t *testing.T){
	s:=&gateBudgetSQLScript{rows:[][]driver.Value{gateBudgetDriverRow(gateBudgetFixture(GateHeld))}}
	p,err:=(SQLGateBudgetReader{DB:openGateBudgetScript(t,s)}).Collect(context.Background(),"installation_primary")
	if err!=nil || p.HoldSeconds!=250 || p.MaxHoldSeconds!=300 || s.begins.Load()!=1 || s.commits.Load()!=1 || s.rollbacks.Load()!=0 {t.Fatalf("original snapshot not committed intact: %+v %v",p,err)}
	if strings.Contains(gateBudgetSnapshotSQL,"lease_expires_at >") || !strings.Contains(gateBudgetSnapshotSQL,"LIMIT 2") || !strings.Contains(gateBudgetSnapshotSQL,"i.active_config_version") {t.Fatal("query can hide expired gate, scan unbounded rows, or use unrelated config")}
}

func TestSQLGateBudgetNeverPublishesMissingPartialFailedOrCanceledSnapshots(t *testing.T){
	row:=gateBudgetDriverRow(gateBudgetFixture(GateHeld))
	for _,name:=range []string{"missing","duplicate","query","read","close","scan","commit","late_cancel"}{t.Run(name,func(t *testing.T){
		s:=&gateBudgetSQLScript{rows:[][]driver.Value{row}};ctx,cancel:=context.WithCancel(context.Background());defer cancel()
		switch name{case "missing":s.rows=nil;case "duplicate":s.rows=[][]driver.Value{row,row};case "query":s.queryErr=errors.New("query rejected");case "read":s.readErr=errors.New("read failed");case "close":s.closeErr=errors.New("close failed");case "scan":bad:=append([]driver.Value(nil),row...);bad[11]=nil;s.rows=[][]driver.Value{bad};case "commit":s.commitErr=errors.New("commit unknown");case "late_cancel":s.onCommit=cancel}
		p,err:=(SQLGateBudgetReader{DB:openGateBudgetScript(t,s)}).Collect(ctx,"installation_primary")
		if err==nil || p!=(GateBudgetProjection{}) {t.Fatalf("partial snapshot published: %+v %v",p,err)}
		if name!="commit" && name!="late_cancel" && s.commits.Load()!=0{t.Fatal("committed failed snapshot")}
	})}
}
