func TestCleanupRecorderCommitsOnlyOriginalAtomicStateAndClock(t *testing.T) {
	s, req, _ := automaticRecordScript(t)
	db := scriptDB(t, s)
	ctx, cancel := context.WithTimeout(context.Background(), time.Minute)
	defer cancel()
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		t.Fatal(err)
	}
	var retained MutationExecutor
	err = RecordInTx(ctx, tx, req, func(executor MutationExecutor, view StartView) error {
		retained = executor
		if len(view.Resources) != 1 || view.Resources[0].Binding.ParentPublicID != req.ParentPublicID {
			t.Fatal("wrong original resource selection")
		}
		// A callback's copy of the view cannot replace the internal plan.
		view.Resources[0].Binding.ResourcePublicID = "srr_00000000-0000-0000-0000-000000000002"
		return executor.Apply()
	})
	if err != nil {
		t.Fatal(err)
	}
	if err = tx.Commit(); err != nil {
		t.Fatal(err)
	}
	if !s.committed || s.rolledBack || len(s.steps) != 0 {
		t.Fatalf("wrong original transaction end: %+v", s)
	}
	before := len(s.seen)
	if retained.Apply() == nil || len(s.seen) != before {
		t.Fatal("retained mutation could execute after its boundary")
	}
}

func TestCleanupRecorderRejectsClockRemovalAndRetiming(t *testing.T) {
	for _, name := range []string{"removed", "start_changed", "deadline_changed", "extra_attempt", "parent_unchanged", "zero_resource_CAS", "callback_error", "panic", "duplicate_Apply"} {
		t.Run(name, func(t *testing.T) {
			s, req, now := automaticRecordScript(t)
			switch name {
			case "removed":
				s.steps[12].values = nil
				s.steps[12].columns = 11
			case "start_changed":
				s.steps[12].values[0][8] = now.Add(-time.Microsecond)
			case "deadline_changed":
				s.steps[12].values[0][9] = now.Add(time.Duration(candidateBudget(t).seconds)*time.Second + time.Second)
			case "extra_attempt":
				extra := append([]driver.Value(nil), s.steps[12].values[0]...)
				extra[0] = uint64(78)
				extra[1] = uint64(2)
				s.steps[12].values = append(s.steps[12].values, extra)
			case "parent_unchanged":
				s.steps[9].values[0][1] = uint64(7)
			case "zero_resource_CAS":
				s.steps[8].affected = 0
			}
			db := scriptDB(t, s)
			ctx, cancel := context.WithTimeout(context.Background(), time.Minute)
			defer cancel()
			tx, err := db.BeginTx(ctx, nil)
			if err != nil {
				t.Fatal(err)
			}
			called := false
			var gotPanic any
			func() {
				defer func() { gotPanic = recover() }()
				err = RecordInTx(ctx, tx, req, func(executor MutationExecutor, _ StartView) error {
					called = true
					if name == "panic" {
						panic("original callback panic")
					}
					if name == "callback_error" {
						return errors.New("original validation failed")
					}
					applyErr := executor.Apply()
					if name == "duplicate_Apply" {
						_ = executor.Apply()
						return nil
					}
					return applyErr
				})
			}()
			if !called {
				t.Fatalf("failed before intended original callback: %v", err)
			}
			if name == "panic" {
				if gotPanic != "original callback panic" {
					t.Fatalf("panic lost: %v", gotPanic)
				}
			} else if err == nil {
				t.Fatal("invalid/corrupted metadata transition accepted")
			}
			if s.committed || !s.rolledBack || !errors.Is(tx.Commit(), sql.ErrTxDone) {
				t.Fatal("ignored failure can commit the original transaction")
			}
		})
	}
}

type dbScript struct {
	steps                 []dbStep
	readOnly              bool
	committed, rolledBack bool
	commitErr             error
	seen                  []string
}
type scriptConnector struct{ s *dbScript }

func (c scriptConnector) Connect(context.Context) (driver.Conn, error) { return &scriptConn{c.s}, nil }
func (c scriptConnector) Driver() driver.Driver                        { return scriptDriver{c.s} }

type scriptDriver struct{ s *dbScript }

func (d scriptDriver) Open(string) (driver.Conn, error) { return &scriptConn{d.s}, nil }

type scriptConn struct{ s *dbScript }

func (*scriptConn) Prepare(string) (driver.Stmt, error) { return nil, errors.New("unexpected prepare") }
func (*scriptConn) Close() error                        { return nil }
func (*scriptConn) Begin() (driver.Tx, error) {
	return nil, errors.New("explicit transaction options required")
}
func (c *scriptConn) BeginTx(_ context.Context, options driver.TxOptions) (driver.Tx, error) {
	if options.ReadOnly != c.s.readOnly || c.s.readOnly && options.Isolation != driver.IsolationLevel(sql.LevelRepeatableRead) {
		return nil, errors.New("wrong transaction boundary")
	}
	return &scriptTx{c.s}, nil
}
func (c *scriptConn) next(query string, exec bool) (dbStep, error) {
	c.s.seen = append(c.s.seen, query)
	if strings.Contains(query, "canonical_config_json") {
		return dbStep{}, errors.New("unknown column in original 066 schema")
	}
	if len(c.s.steps) == 0 {
		return dbStep{}, fmt.Errorf("unexpected SQL %s", query)
	}
	step := c.s.steps[0]
	c.s.steps = c.s.steps[1:]
	if step.exec != exec || !strings.Contains(query, step.match) {
		return dbStep{}, fmt.Errorf("SQL order/statement mismatch: want %s got %s", step.match, query)
	}
	return step, step.err
}
func (c *scriptConn) QueryContext(_ context.Context, query string, _ []driver.NamedValue) (driver.Rows, error) {
	step, err := c.next(query, false)
	if err != nil {
		return nil, err
	}
	return &scriptRows{step: step}, nil
}
func (c *scriptConn) ExecContext(_ context.Context, query string, _ []driver.NamedValue) (driver.Result, error) {
	step, err := c.next(query, true)
	if err != nil {
		return nil, err
	}
	return driver.RowsAffected(step.affected), nil
}

type scriptTx struct{ s *dbScript }

func (t *scriptTx) Commit() error   { t.s.committed = true; return t.s.commitErr }
func (t *scriptTx) Rollback() error { t.s.rolledBack = true; return nil }

type scriptRows struct {
	step  dbStep
	index int
}

func (r *scriptRows) Columns() []string {
	n := r.step.columns
	if n == 0 && len(r.step.values) > 0 {
		n = len(r.step.values[0])
	}
	names := make([]string, n)
	for i := range names {
		names[i] = fmt.Sprintf("column_%d", i)
	}
	return names
}
func (*scriptRows) Close() error { return nil }
func (r *scriptRows) Next(dest []driver.Value) error {
	if r.index >= len(r.step.values) {
		if r.step.rowsErr != nil {
			err := r.step.rowsErr
			r.step.rowsErr = nil
			return err
		}
		return io.EOF
	}
	copy(dest, r.step.values[r.index])
	r.index++
	return nil
}
func scriptDB(t *testing.T, s *dbScript) *sql.DB {
	t.Helper()
	db := sql.OpenDB(scriptConnector{s})
	db.SetMaxOpenConns(1)
	t.Cleanup(func() { db.Close() })
	return db
}

func clockStep(now time.Time) dbStep {
	return dbStep{match: "SELECT UTC_TIMESTAMP(6),", values: [][]driver.Value{{now, int64(0)}}}
}
func emptyParents(kind string) dbStep {
	match := "SELECT DISTINCT b.public_id"
	if kind == "drill" {
		match = "SELECT public_id FROM system_restore_drills"
	}
	return dbStep{match: match, columns: 1}
}
func parentStep(t *testing.T, finished time.Time) dbStep {
	t.Helper()
	raw := candidateCanonical(t)
	budget := candidateBudget(t)
	var finish driver.Value
	if !finished.IsZero() {
		finish = finished
	}
	return dbStep{match: "JOIN system_backup_configs", values: [][]driver.Value{{uint64(3), raw, int64(len(raw)), budget.sha256, finish, budget.sha256, int64(len(raw)), budget.seconds}}}
}

type residualModel struct {
	value    *bool
	misbound bool
	err      error
	calls    int
}

func (m *residualModel) ReadPersisted(_ context.Context, _ *sql.Tx, identity ResourceIdentity, _ time.Time) (ResidualObservation, error) {
	m.calls++
	if m.misbound {
		identity.LocationSHA256 = strings.Repeat("b", 64)
	}
	return ResidualObservation{identity, m.value}, m.err
}

func TestCleanupReaderRetainsFailedAttemptResidualAfterNewRetry(t *testing.T) {
	now := time.Date(2026, 10, 27, 1, 2, 3, 0, time.UTC)
	binding := candidateBinding("drill")
	budget := candidateBudget(t)
	start := now.Add(-time.Duration(budget.seconds)*time.Second - time.Second)
	for _, name := range []string{"residual_true", "settled_medium", "unknown_medium", "misbound_medium", "foreign_operation", "history_read_error", "missing_history"} {
		t.Run(name, func(t *testing.T) {
			resource := []driver.Value{binding.ResourcePublicID, "namespace", "deleted", "namespace-uid", nil, strings.Repeat("a", 64), strings.Repeat("c", 64), uint64(9)}
			old := []driver.Value{uint64(1), string(AutomaticDrill), uint64(3), budget.sha256, start, start.Add(time.Duration(budget.seconds) * time.Second), start, nil, nil, nil, nil, nil, nil, nil, nil, nil}
			opStart := now.Add(-10 * time.Second)
			install := binding.Installation
			if name == "foreign_operation" {
				install = "installation_foreign"
			}
			current := []driver.Value{uint64(2), string(ExplicitOperation), uint64(3), budget.sha256, opStart, opStart.Add(60 * time.Second), opStart, uint64(9), uint64(9), install, "sop_00000000-0000-0000-0000-000000000001", "cleanup", "drill", binding.ParentPublicID, opStart, opStart.Add(60 * time.Second)}
			truth := name != "settled_medium"
			observer := &residualModel{value: &truth, misbound: name == "misbound_medium"}
			if name == "unknown_medium" {
				observer.value = nil
			}
			s := &dbScript{readOnly: true, steps: []dbStep{clockStep(now), emptyParents("backup"), {match: "SELECT public_id FROM system_restore_drills", values: [][]driver.Value{{binding.ParentPublicID}}}, parentStep(t, start), {match: "FROM system_restore_drill_resources", values: [][]driver.Value{resource}}}}
			if name != "unknown_medium" && name != "misbound_medium" {
				history := dbStep{match: "FROM system_backup_cleanup_attempts", values: [][]driver.Value{old, current}}
				if name == "history_read_error" {
					history.rowsErr = errors.New("late history error")
				}
				if name == "missing_history" {
					history.values = nil
					history.columns = 16
				}
				s.steps = append(s.steps, history)
			}
			collection, err := (Reader{DB: scriptDB(t, s), Residual: observer}).Collect(context.Background(), binding.Installation)
			if name == "residual_true" {
				if err != nil || collection == nil || len(collection.Owners) != 2 || collection.Owners[0].Age <= collection.Owners[0].ImmutableDeadline || !collection.Owners[0].ResidualVolume || collection.Owners[1].Age >= collection.Owners[1].ImmutableDeadline {
					t.Fatalf("failed old medium disappeared on retry: %+v %v", collection, err)
				}
			} else if name == "settled_medium" {
				if err != nil || collection == nil || len(collection.Owners) != 0 {
					t.Fatalf("verified settlement did not recover: %+v %v", collection, err)
				}
			} else if err == nil || collection != nil || s.committed {
				t.Fatalf("unknown/partial/misbound facts published: %+v %v", collection, err)
			}
			if len(s.steps) != 0 {
				t.Fatalf("failed before the intended observation: %d remaining", len(s.steps))
			}
		})
	}
}
